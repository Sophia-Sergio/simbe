## Global Constraints
- Secrets only in GitHub/Fly secret stores; `FLY_API_TOKEN` lives in the GitHub **environment `production`**. No secret, export, `.env` or key in git (the `guards` job enforces it).
- **The Fly machine must stay up while background jobs run**: imports run as Solid Queue jobs inside the Puma process, so `auto_stop_machines = "off"`, `min_machines_running = 1` (appketing scales to zero; Simbe must not).
- The deploy job runs only on `main`, only after every required job passed, and never in parallel with itself.
- Workflows pass user-controlled values (branch names, PR titles) through `env:`, never inline in `run:` (actionlint enforces this).
- **`mockServiceWorker.js` and the MSW/mocks code never ship**: the MSW worker lives only in `app/javascript/mock-host/public/` and mocks are loaded behind a build-time `__MOCKS_ENABLED__` flag that is `false` in every Rails build.

## Review Focus
- A PR from a `lane/*` branch that edits an unowned file fails `ownership_check`; a `ccr`-labelled PR touching only `contracts/**` passes.
- `docker build` works from the repo root with no secrets and produces `public/vite/**`; the image serves `/up`, the SPA shell at `/` and JSON `401` at `/api/v1/me`.
- `e2e_real` never needs the real exports (never in CI): it uploads the synthetic export through the browser.
- Losing `AR_ENCRYPTION_*` keys makes screener answers unreadable — the runbook makes backing them up a blocking step.

```dockerfile
RUN apt-get update -qq && \
    apt-get install --no-install-recommends -y curl libjemalloc2 libpq5 postgresql-client && \

    LD_PRELOAD="/usr/local/lib/libjemalloc.so" RAILS_SERVE_STATIC_FILES="true" RAILS_LOG_TO_STDOUT="true" SOLID_QUEUE_IN_PUMA="true"

FROM base AS build
RUN apt-get update -qq && apt-get install --no-install-recommends -y build-essential git libpq-dev libyaml-dev pkg-config && \
    rm -rf /var/lib/apt/lists /var/cache/apt/archives
COPY --from=docker.io/library/node:22-bookworm-slim /usr/local/bin/node /usr/local/bin/node
COPY --from=docker.io/library/node:22-bookworm-slim /usr/local/lib/node_modules /usr/local/lib/node_modules
RUN ln -sf /usr/local/lib/node_modules/npm/bin/npm-cli.js /usr/local/bin/npm && ln -sf /usr/local/lib/node_modules/npm/bin/npx-cli.js /usr/local/bin/npx
COPY .ruby-version Gemfile Gemfile.lock ./
RUN bundle install && rm -rf ~/.bundle/ "${BUNDLE_PATH}"/ruby/*/cache "${BUNDLE_PATH}"/ruby/*/bundler/gems/*/.git && \
    bundle exec bootsnap precompile --gemfile
COPY package.json package-lock.json ./
RUN npm ci
COPY . .
RUN chmod +x bin/fly-start bin/docker-entrypoint && bundle exec bootsnap precompile app/ lib/
# vite_rails hooks `assets:precompile` into the Vite build (output: public/vite). No real secrets are needed to build.
RUN SECRET_KEY_BASE_DUMMY=1 DATABASE_URL=postgres://build:build@127.0.0.1/build \
    AR_ENCRYPTION_PRIMARY_KEY=build AR_ENCRYPTION_DETERMINISTIC_KEY=build AR_ENCRYPTION_KEY_DERIVATION_SALT=build \
    ./bin/rails assets:precompile && rm -rf node_modules

FROM base
RUN groupadd --system --gid 1000 rails && useradd rails --uid 1000 --gid 1000 --create-home --shell /bin/bash
USER 1000:1000
COPY --chown=rails:rails --from=build "${BUNDLE_PATH}" "${BUNDLE_PATH}"
COPY --chown=rails:rails --from=build /rails /rails
ENTRYPOINT ["/rails/bin/docker-entrypoint"]
EXPOSE 8080
CMD ["./bin/fly-start"]
```

## Task 3: The CI workflow

**Files:** `.github/workflows/ci.yml` (replaces W0-1's minimal workflow).
- [ ] **Step 1:** Write this workflow **verbatim** (already validated with `actionlint` and PyYAML while planning; re-validate after any edit). Jobs: `decide`, `contract`, `brakeman`, `bundler_audit`, `npm_audit`, `knip`, `lint_backend`, `lint_frontend`, `test_backend`, `test_frontend`, `e2e_mock`, `e2e_real`, `guards`, `ownership_check`, `docker_build`, `deploy`.
```yaml
name: CI

on:
  pull_request:
  push:
    branches: [ main ]
  schedule:
    - cron: "0 7 * * *"   # nightly e2e_real

permissions:
  contents: read
  actions: read
  pull-requests: read

concurrency:
  group: ci-${{ github.event.pull_request.number || github.ref }}
  cancel-in-progress: ${{ github.event_name == 'pull_request' }}

env:
  NODE_VERSION: "22"
  # Non-secret dummy keys so Active Record Encryption works in CI (never use these anywhere else).
  AR_ENCRYPTION_PRIMARY_KEY: ci-primary-key-0123456789abcdef0123456789abcdef
  AR_ENCRYPTION_DETERMINISTIC_KEY: ci-deterministic-key-0123456789abcdef01234567
  AR_ENCRYPTION_KEY_DERIVATION_SALT: ci-salt-0123456789abcdef0123456789abcdef

jobs:
  decide:
    name: Decide whether CI already ran
    runs-on: ubuntu-latest
    outputs:
      run_ci: ${{ steps.decide.outputs.run_ci }}
    steps:
      - name: Skip CI when this commit already passed on a merged PR
        id: decide
        uses: actions/github-script@v7
        with:
          script: |
            if (context.eventName !== 'push' || context.ref !== 'refs/heads/main') {
              core.setOutput('run_ci', 'true'); return
            }
            const { data: prs } = await github.rest.repos.listPullRequestsAssociatedWithCommit({
              owner: context.repo.owner, repo: context.repo.repo, commit_sha: context.sha })
            const merged = prs.find((pr) => pr.merged_at)
            if (!merged) { core.setOutput('run_ci', 'true'); return }
            const { data: runs } = await github.rest.actions.listWorkflowRuns({
              owner: context.repo.owner, repo: context.repo.repo, workflow_id: 'ci.yml', head_sha: merged.head.sha, per_page: 20 })
            const passed = runs.workflow_runs.some((r) => r.event === 'pull_request' && r.conclusion === 'success')
            core.setOutput('run_ci', passed ? 'false' : 'true')

  contract:
    name: Contract
    needs: decide
    if: needs.decide.outputs.run_ci == 'true'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: "${{ env.NODE_VERSION }}", cache: npm, cache-dependency-path: package-lock.json }
      - run: npm ci
      - name: Bundle and lint the OpenAPI contract (validates every example)   # parity
        run: npm run contract
      - name: Generated API types are up to date   # parity
        run: npm run types && git diff --exit-code src/api/generated
      - name: No two lanes own the same files   # parity
        run: bash docs/superpowers/parallel/check-ownership.sh

  brakeman:
    name: Brakeman
    needs: decide
    if: needs.decide.outputs.run_ci == 'true'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: ruby/setup-ruby@v1
        with: { bundler-cache: true }
      - run: bundle exec brakeman -q --no-pager   # parity

  bundler_audit:
    name: Bundler audit
    needs: decide
    if: needs.decide.outputs.run_ci == 'true'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: ruby/setup-ruby@v1
        with: { bundler-cache: true }
      - run: bundle exec bundler-audit check --update   # parity

  npm_audit:
    name: npm audit
    needs: decide
    if: needs.decide.outputs.run_ci == 'true'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: "${{ env.NODE_VERSION }}" }
      - run: npm audit --audit-level=high   # parity

  knip:
    name: Unused frontend code
    needs: decide
    if: needs.decide.outputs.run_ci == 'true'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: "${{ env.NODE_VERSION }}", cache: npm, cache-dependency-path: package-lock.json }
      - run: npm ci
      - run: npx knip   # parity

  lint_backend:
    name: Lint backend
    needs: decide
    if: needs.decide.outputs.run_ci == 'true'
    runs-on: ubuntu-latest
    env:
      RUBOCOP_CACHE_ROOT: tmp/rubocop
    steps:
      - uses: actions/checkout@v4
      - uses: ruby/setup-ruby@v1
        with: { bundler-cache: true }
      - uses: actions/cache@v4
        with:
          path: tmp/rubocop
          key: rubocop-${{ runner.os }}-${{ hashFiles('.ruby-version', '.rubocop.yml', 'Gemfile.lock') }}
      - run: bundle exec rubocop --format github --format simple   # parity
      - name: Architecture boundaries
        env: { DATABASE_URL: "postgres://unused:unused@127.0.0.1/unused" }
        run: bundle exec rails boundaries:check   # parity

  lint_frontend:
    name: Lint frontend
    needs: decide
    if: needs.decide.outputs.run_ci == 'true'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: "${{ env.NODE_VERSION }}", cache: npm, cache-dependency-path: package-lock.json }
      - run: npm ci
      - run: npm run lint   # parity
      - run: npm run typecheck   # parity

  test_backend:
    name: Backend tests
    needs: decide
    if: needs.decide.outputs.run_ci == 'true'
    runs-on: ubuntu-latest
    services:
      postgres:
        image: postgres:16
        env: { POSTGRES_USER: postgres, POSTGRES_PASSWORD: postgres }
        ports: [ "5432:5432" ]
        options: --health-cmd="pg_isready -U postgres" --health-interval=10s --health-timeout=5s --health-retries=5
    steps:
      - run: sudo apt-get update && sudo apt-get install --no-install-recommends -y libpq-dev
      - uses: actions/checkout@v4
      - uses: ruby/setup-ruby@v1
        with: { bundler-cache: true }
      - uses: actions/setup-node@v4
        with: { node-version: "${{ env.NODE_VERSION }}", cache: npm, cache-dependency-path: package-lock.json }
      - name: Install JS deps (Vite builds the test assets) and bundle the OpenAPI contract for conformance checks
        run: npm ci && npm run contract
      - name: RSpec (OpenAPI conformance on)   # parity
        env:
          RAILS_ENV: test
          DATABASE_URL: postgres://postgres:postgres@localhost:5432
        run: |
          bundle exec rake 'parallel:create[2]' 'parallel:load_schema[2]'
          bundle exec parallel_rspec spec -n 2 --group-by runtime --allowed-missing 100
          bundle exec rake contract:coverage | tee contract-coverage.txt
      - uses: actions/upload-artifact@v4
        if: always()
        with: { name: contract-coverage, path: contract-coverage.txt, if-no-files-found: ignore }

  test_frontend:
    name: Frontend tests
    needs: decide
    if: needs.decide.outputs.run_ci == 'true'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: "${{ env.NODE_VERSION }}", cache: npm, cache-dependency-path: package-lock.json }
      - run: npm ci
      - name: Vitest (includes mock-conformance against the contract)   # parity
        run: npm test

  e2e_mock:
    name: E2E (mock API, desktop + mobile)
    needs: decide
    if: needs.decide.outputs.run_ci == 'true'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: "${{ env.NODE_VERSION }}", cache: npm, cache-dependency-path: package-lock.json }
      - run: npm ci && npx playwright install --with-deps chromium
      - name: Start the standalone mock host (no Rails, no Docker)
        run: |
          npm run dev:mock &
          npx --yes wait-on http://127.0.0.1:5173 --timeout 60000
      - name: Playwright (desktop + mobile projects)   # parity
        env: { E2E_BASE_URL: "http://127.0.0.1:5173" }
        run: npx playwright test
      - uses: actions/upload-artifact@v4
        if: always()
        with:
          name: e2e-mock-screenshots
          path: |
            e2e/screenshots
            test-results
          if-no-files-found: ignore

  e2e_real:
    name: E2E (real API)
    needs: decide
    # main, nightly, or PRs labelled `e2e`; not required until Wave 3 (W3-1) flips continue-on-error off
    if: >
      needs.decide.outputs.run_ci == 'true' &&
      (github.event_name == 'schedule' || github.ref == 'refs/heads/main' || contains(github.event.pull_request.labels.*.name, 'e2e'))
    runs-on: ubuntu-latest
    continue-on-error: true
    services:
      postgres:
        image: postgres:16
        env: { POSTGRES_USER: postgres, POSTGRES_PASSWORD: postgres }
        ports: [ "5432:5432" ]
        options: --health-cmd="pg_isready -U postgres" --health-interval=10s --health-timeout=5s --health-retries=5
    env:
      RAILS_ENV: development
      DATABASE_URL: postgres://postgres:postgres@localhost:5432/simbe_e2e
      E2E_STUB_ANTHROPIC: "1"
      E2E_API: real
    steps:
      - run: sudo apt-get update && sudo apt-get install --no-install-recommends -y libpq-dev
      - uses: actions/checkout@v4
      - uses: ruby/setup-ruby@v1
        with: { bundler-cache: true }
      - uses: actions/setup-node@v4
        with: { node-version: "${{ env.NODE_VERSION }}", cache: npm, cache-dependency-path: package-lock.json }
      - run: npm ci && npx playwright install --with-deps chromium
      - name: Prepare database and seed the Alpha dataset
        run: bundle exec rails db:create db:schema:load db:seed e2e:seed
      - name: Generate a synthetic Harbor-shaped export (3,500 leads) for the upload-at-signup journey
        run: python3 bin/generate-sample-export --out e2e/fixtures/synthetic --leads 3500 --seed 7
      - name: Build assets and start Rails (Solid Queue in Puma serves the SPA and the API)
        run: |
          bin/vite build
          SOLID_QUEUE_IN_PUMA=true bundle exec rails server -b 127.0.0.1 -p 3000 &
          npx --yes wait-on http://127.0.0.1:3000/up --timeout 90000
      - name: "Playwright against the real API (real-exports journeys are skipped - the exports are never in CI)"
        env: { E2E_BASE_URL: "http://127.0.0.1:3000", E2E_SYNTHETIC_DIR: "e2e/fixtures/synthetic" }
        run: npx playwright test --grep-invert @real
      - uses: actions/upload-artifact@v4
        if: always()
        with: { name: e2e-real-screenshots, path: "e2e/screenshots\ntest-results", if-no-files-found: ignore }

  guards:
    name: Guards (no health data or secrets committed)
    needs: decide
    if: needs.decide.outputs.run_ci == 'true'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with: { fetch-depth: 0 }
      - name: No exports, env files or stray CSVs are tracked   # parity
        run: bin/guard-no-health-data
      - name: Golden-numbers oracle is valid JSON and lane tooling loads   # parity
        run: |
          python3 -c "import json; json.load(open('spec/real_data/golden.json'))"
          bin/lane list > /dev/null
      - name: Scan for secrets
        uses: gitleaks/gitleaks-action@v2
        env: { GITHUB_TOKEN: "${{ secrets.GITHUB_TOKEN }}" }

  ownership_check:
    name: Lane ownership
    needs: decide
    if: github.event_name == 'pull_request' && startsWith(github.head_ref, 'lane/')
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with: { fetch-depth: 0 }
      - name: Changed files are inside the lane's Owns list
        env:
          PR_LABELS: ${{ join(github.event.pull_request.labels.*.name, ' ') }}
          HEAD_REF: ${{ github.head_ref }}
          BASE_REF: ${{ github.base_ref }}
        run: |
          lane="$(echo "$HEAD_REF" | sed -E 's#^lane/([A-Za-z0-9-]+)-[a-z0-9].*#\1#')"
```
