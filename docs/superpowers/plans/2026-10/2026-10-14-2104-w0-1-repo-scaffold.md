```ts
import type { FeatureModule, NavItem, Role } from "./feature";

export function buildRegistry(features: FeatureModule[]) {
  const ids = new Set<string>();
  for (const f of features) {
    if (ids.has(f.id)) throw new Error(`Duplicate feature id: ${f.id}`);
    ids.add(f.id);
  }
  const navAll = features.flatMap((f) => f.nav ?? []);
  return {
    features,
    routes: features.flatMap((f) => f.routes ?? []),
    publicRoutes: features.flatMap((f) => f.publicRoutes ?? []),
    nav: (role: Role): NavItem[] => navAll.filter((n) => n.roles.includes(role)).sort((x, y) => x.order - y.order),
    overlays: features.flatMap((f) => f.overlays ?? []),
    providers: features.flatMap((f) => f.providers ?? []).sort((x, y) => x.order - y.order),
    fabs: features.flatMap((f) => (f.fab ? [{ id: f.id, component: f.fab }] : [])),
    topBarRight: features.flatMap((f) => f.topBarRight ?? []),
  };
}

const modules = import.meta.glob<{ feature: FeatureModule }>("/app/javascript/features/*/feature.ts", { eager: true });
export const registry = buildRegistry(Object.values(modules).map((m) => m.feature));
```
