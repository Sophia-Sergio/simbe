```ts
import type { FeatureModule, NavItem, Role } from "./feature";

export function buildRegistry(features: FeatureModule[]) {
  const ids = new Set<string>();
  for (const f of features) {
    if (ids.has(f.id)) throw new Error(`Duplicate feature id: ${f.id}`);
    ids.add(f.id);
  }
  const navAll = features.flatMap((f) => f.nav ?? []);
  return {
    features,
    routes: features.flatMap((f) => f.routes ?? []),
    publicRoutes: features.flatMap((f) => f.publicRoutes ?? []),
    nav: (role: Role): NavItem[] => navAll.filter((n) => n.roles.includes(role)).sort((x, y) => x.order - y.order),
    overlays: features.flatMap((f) => f.overlays ?? []),
    providers: features.flatMap((f) => f.providers ?? []).sort((x, y) => x.order - y.order),
    fabs: features.flatMap((f) => (f.fab ? [{ id: f.id, component: f.fab }] : [])),
    topBarRight: features.flatMap((f) => f.topBarRight ?? []),
  };
}

const modules = import.meta.glob<{ feature: FeatureModule }>("/src/features/*/feature.ts", { eager: true });
export const registry = buildRegistry(Object.values(modules).map((m) => m.feature));
```

```bash
#!/usr/bin/env bash
# Usage: pr-ownership.sh <lane-id> [base-ref]   e.g. pr-ownership.sh F-PIPELINE origin/main
set -euo pipefail
lane="$1"; base="${2:-origin/main}"
plan=$(grep -l "^\*\*Lane:\*\* ${lane}\b" docs/superpowers/plans/2026-10/*.md | head -1) || true
[ -n "$plan" ] || { echo "no plan declares lane ${lane}"; exit 2; }
globs=$(grep -m1 '^\*\*Owns:\*\*' "$plan" | grep -o '`[^`]*`' | tr -d '`')
changed=$(git diff --name-only "$base"...HEAD)
labels="${PR_LABELS:-}"
fail=0
while IFS= read -r f; do
  [ -z "$f" ] && continue
  ok=0
  while IFS= read -r g; do
    # shellcheck disable=SC2254
    case "$f" in $(printf '%s' "$g" | sed 's#\*\*#*#g')) ok=1;; esac
  done <<< "$globs"
  [[ "$f" == docs/superpowers/plans/* ]] && ok=1
  [[ "$labels" == *ccr* && ( "$f" == contracts/* || "$f" == frontend/src/api/generated/* ) ]] && ok=1
  [ $ok -eq 1 ] || { echo "OUTSIDE LANE ${lane}: $f"; fail=1; }
done <<< "$changed"
exit $fail
```
