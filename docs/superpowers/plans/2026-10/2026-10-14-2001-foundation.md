# Foundation Implementation Plan

> **For Cursor / OpenCode:** Implement task-by-task. Follow `docs/conventions/`. Stop if blocked — do not replan silently. Steps use checkbox (`- [ ]`) syntax. **Commit after every task** (message: `feat(foundation): task N – <title>`), after `make verify` (or the task's own checks while `verify` is still being assembled) passes.

**Goal:** A Dockerized Rails API + Postgres + React shell with sign-in, owner/coordinator roles, and enforced agency isolation.

**Spec:** `docs/superpowers/specs/2026-10/2026-10-14-2000-foundation-design.md`

**Depends on:** nothing (first plan). **Unblocks:** all other plans.

**Architecture:** Rails 8 API-only app in `backend/` (cookie-session auth, `Current.agency`, `AgencyScoped` concern); Vite React TS SPA in `frontend/` talking to `/api/v1` through one `jsonRequest` layer; root docker-compose + Makefile drive both.

**Tech Stack:** Ruby 3.3, Rails 8.0, PostgreSQL 16, RSpec, FactoryBot, WebMock, RuboCop omakase, Brakeman, bundler-audit · Node 20, React 18, TypeScript, Vite 5, Tailwind 3.4, TanStack Query 5, React Router 6, lucide-react, Vitest, Testing Library, Playwright.

## Global Constraints
- Breakpoint **860px**; layout per `design_handoff_simbe/README.md`; **colors, fonts, radii, shadows and logo exactly as in `docs/design/brand.md`** (simbefamily.com brand — it overrides the handoff's brand tokens).
- Other agencies' ids → **404**. Never take `agency_id` from params.
- Commands run through Docker/`make` only.

## Review Focus
- Session cookie expired/forged/missing → 401 `unauthenticated`, never 500.
- Non-GET without `X-Requested-With: simbe` → 403 `csrf`.
- Coordinator hitting an owner-only route → 403 `forbidden`.
- User with deleted agency / nil `Current.agency` → raises, never silently unscoped.
- Window resize across 860px switches shell without losing the route.

`app/models/auth_session.rb`:
```ruby
class AuthSession < ApplicationRecord
  TTL = 14.days
  belongs_to :user
  def self.issue!(user)
    token = SecureRandom.urlsafe_base64(32)
    create!(user: user, token_digest: digest(token), expires_at: TTL.from_now)
    token
  end
  def self.find_by_token(token)
    return if token.blank?
    where("expires_at > ?", Time.current).find_by(token_digest: digest(token))
  end
  def self.digest(token) = OpenSSL::Digest::SHA256.hexdigest(token)
end
```
`app/controllers/api/base_controller.rb`:
```ruby
module Api
  class BaseController < ActionController::API
    include ActionController::Cookies
    before_action :require_csrf_header!
    before_action :authenticate!
    after_action { Current.reset }

    rescue_from ActiveRecord::RecordNotFound do render_error("not_found", "Not found", :not_found) end
    rescue_from Agency::MissingContext do render_error("unauthenticated", "Sign in required", :unauthorized) end

    private

    def require_csrf_header!
      return if request.get? || request.head?
      return if request.headers["X-Requested-With"] == "simbe"
      render_error("csrf", "Missing X-Requested-With header", :forbidden)
    end

    def authenticate!
      session = AuthSession.find_by_token(cookies.signed[:simbe_session])
      return render_error("unauthenticated", "Sign in required", :unauthorized) unless session
      Current.user = session.user
      Current.agency = session.user.agency
    end

    def require_owner!
      render_error("forbidden", "Owner access required", :forbidden) unless Current.user.owner?
    end

    def render_error(code, message, status)
      render json: { error: { code: code, message: message } }, status: status
    end
  end
end
```
`app/controllers/api/v1/sessions_controller.rb`:

```ruby
module Api
  module V1
    class SessionsController < Api::BaseController
      skip_before_action :authenticate!, only: :create
      def create
        user = User.find_by(email: params[:email].to_s.strip.downcase)
        unless user&.authenticate(params[:password].to_s)
          return render_error("invalid_credentials", "Email or password is incorrect", :unauthorized)
        end
        token = AuthSession.issue!(user)
        cookies.signed[:simbe_session] = { value: token, httponly: true, same_site: :lax,
                                           secure: Rails.env.production?, expires: AuthSession::TTL.from_now }
        render json: MeSerializer.call(user), status: :created
      end

      def destroy
        AuthSession.find_by_token(cookies.signed[:simbe_session])&.destroy
        cookies.delete(:simbe_session)
        head :no_content
      end
    end
  end
end
```

```ruby
namespace :boundaries do
  desc "Static architecture rules (allowlist only shrinks)"
  task check: :environment do
    rules = {
      "unscoped is only allowed in Snapshot services and specs" =>
        [ /\.unscoped\b/, %w[app lib], %r{\Aapp/services/snapshot/} ],
      "Anthropic::Client only inside Ai::Client" =>
        [ /Anthropic::Client/, %w[app lib], %r{\Aapp/services/ai/client\.rb\z} ],
      "no raw `fetch` of render json: record" =>
        [ /render json: (?!\{|\w*Serializer|MeSerializer)\w+\s*$/, %w[app/controllers], nil ]
    }
    failures = []
    rules.each do |name, (regex, dirs, allow)|
      dirs.each do |d|
        Dir[Rails.root.join(d, "**/*.rb")].each do |f|
          rel = Pathname(f).relative_path_from(Rails.root).to_s
          next if allow && rel.match?(allow)
          File.foreach(f).with_index(1) { |l, i| failures << "#{name}: #{rel}:#{i}" if l.match?(regex) }
        end
      end
    end
    abort(failures.join("\n")) if failures.any?
    puts "boundaries ok"
  end
end
```

```ts
export class ApiError extends Error {
  constructor(public status: number, public code: string, message: string) { super(message); }
}
export async function jsonRequest<T = unknown>(method: string, path: string, body?: unknown): Promise<T> {
  const res = await fetch(path, {
    method, credentials: "same-origin",
    headers: { Accept: "application/json", ...(body !== undefined ? { "Content-Type": "application/json" } : {}), "X-Requested-With": "simbe" },
    body: body !== undefined ? JSON.stringify(body) : undefined,
  });
  if (res.status === 204) return undefined as T;
  const data = await res.json().catch(() => ({}));
  if (!res.ok) throw new ApiError(res.status, data?.error?.code ?? "unknown", data?.error?.message ?? res.statusText);
  return data as T;
}
```

```ts
export function useAsyncAction<A extends unknown[], R>(fn: (...a: A) => Promise<R>) {
  const [pending, setPending] = useState(false);
  const inFlight = useRef(false);
  const run = useCallback(async (...args: A) => {
    if (inFlight.current) return undefined;
    inFlight.current = true; setPending(true);
    try { return await fn(...args); } finally { inFlight.current = false; setPending(false); }
  }, [fn]);
  return { run, pending };
}
```
